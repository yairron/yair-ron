4.2 פונקציית getVisibleFields(state)
הפונקציה המרכזית. מחזירה מערך של שמות שדות שצריכים להיות גלויים לפי ה-state הנוכחי:
function getVisibleFields(state) {
const fields = ["gender", "birthDate", "checkType"];
if (!state.gender) return fields;
if (state.checkType === "full") fields.push("toshavHozer");
// מצב משפחתי לפי מין
if (state.gender === "male") {
fields.push("maritalStatus_Male");
if (state.maritalStatus === 1) {
fields.push("partnerResident");
if (state.partnerResident === true) fields.push("marriedTime");
}
} else {
fields.push("maritalStatus_Female");
if (state.maritalStatus === 1) fields.push("partnerResidentFemale");
if (state.maritalStatus === 0) fields.push("toshevetHozeret");
if (state.maritalStatus === 3) fields.push("widowKitsba");
}
// תלויים — תמיד
fields.push("TxtWith1","TxtWith2","TxtWith3","TxtWith4","TxtWith5");
// תקופת אכשרה — מלאה בלבד
if (state.checkType === "full") fields.push("qualificationPeriod");
// ותק + נכות
fields.push("vetek", "disabilityPension");
if (state.disabilityPension === false) {
fields.push("incomeNotFromWork", "doYouWork");
if (state.doYouWork === true) fields.push("workType");
}
return fields;
}

function isComplete(state) {
const required = getVisibleFields(state).filter(f => REQUIRED_FIELDS.includes(f));
return required.every(f => state[f] !== null && state[f] !== undefined);
